cmake_minimum_required(VERSION 3.28)
project(tower_native_settings LANGUAGES CXX)
find_package(ERNativeUI CONFIG REQUIRED)
add_library(tower_native_settings SHARED settings.cpp)
target_link_libraries(tower_native_settings PRIVATE ERNativeUI::SDK)
target_compile_features(tower_native_settings PRIVATE cxx_std_17)
target_compile_definitions(tower_native_settings PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(tower_native_settings PRIVATE /W4 /permissive- /utf-8)
set_property(TARGET tower_native_settings PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
